    // 遊戲狀態變數

    let clearedDepts = new Set();

    let score = 0;

    let currentDeptIndex = null;

    let audioCtx = null;

 

    // Web Audio API：動態生成答對/答錯音效

    function getAudioContext() {

      if (!audioCtx) {

        audioCtx = new (window.AudioContext || window.webkitAudioContext)();

      }

      return audioCtx;

    }

 

    // 答對音效

    function playCorrectSound() {

      try {

        const ctx = getAudioContext();

        const now = ctx.currentTime;

        const osc = ctx.createOscillator();

        const gain = ctx.createGain();

 

        osc.type = 'sine';

        osc.frequency.setValueAtTime(523.25, now);

        osc.frequency.exponentialRampToValueAtTime(1046.50, now + 0.3);

 

        gain.gain.setValueAtTime(0.3, now);

        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);

 

        osc.connect(gain);

        gain.connect(ctx.destination);

 

        osc.start(now);

        osc.stop(now + 0.4);

      } catch (e) { console.log(e); }

    }

 

    // 答錯音效

    function playWrongSound() {

      try {

        const ctx = getAudioContext();

        const now = ctx.currentTime;

        const osc = ctx.createOscillator();

        const gain = ctx.createGain();

 

        osc.type = 'triangle';

        osc.frequency.setValueAtTime(160, now);

        osc.frequency.exponentialRampToValueAtTime(80, now + 0.2);

 

        gain.gain.setValueAtTime(0.4, now);

        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);

 

        osc.connect(gain);

        gain.connect(ctx.destination);

 

        osc.start(now);

        osc.stop(now + 0.2);

      } catch (e) { console.log(e); }

    }

 

    // 切換音樂播放狀態

    function toggleBGM() {

      const bgm = document.getElementById("bgm");

      const btn = document.getElementById("bgm-btn");

      if (bgm.paused) {

        bgm.play();

        btn.innerText = "⏸️ 暫停背景音樂";

      } else {

        bgm.pause();

        btn.innerText = "🎵 播放背景音樂";

      }

    }

 

    // 通關金粉慶祝特效

    function triggerGoldShower() {

      var duration = 4 * 1000;

      var end = Date.now() + duration;

 

      (function frame() {

        confetti({

          particleCount: 7,

          angle: 60,

          spread: 55,

          origin: { x: 0 },

          colors: ['#d4af37', '#ffd700', '#8b0000', '#ffffff']

        });

        confetti({

          particleCount: 7,

          angle: 120,

          spread: 55,

          origin: { x: 1 },

          colors: ['#d4af37', '#ffd700', '#8b0000', '#ffffff']

        });

 

        if (Date.now() < end) {

          requestAnimationFrame(frame);

        }

      })();

    }

 

    // 初始化網頁與卡片

    function initGame() {

      const grid = document.getElementById("dept-grid");

      grid.innerHTML = "";

 

      departmentsData.forEach((dept, index) => {

        const card = document.createElement("div");

        card.className = "dept-card";

        card.id = `card-${index}`;

        card.onclick = () => openModal(index);

 

        card.innerHTML = `

          <div class="dept-number">第 ${index + 1} 司</div>

          <div class="dept-name">${dept.name}</div>

        `;

        grid.appendChild(card);

      });

 

      updateStats();

    }

 

    // 更新統計資訊

    function updateStats() {

      document.getElementById("cleared-count").innerText = clearedDepts.size;

      document.getElementById("score").innerText = score;

    }

 

    // 打開關卡 Modal

    function openModal(index) {

      currentDeptIndex = index;

      const dept = departmentsData[index];

 

      document.getElementById("modal-dept-name").innerText = `${dept.name}（${dept.desc}）`;

      document.getElementById("modal-dialogue").innerText = dept.dialogue;

      document.getElementById("modal-question").innerText = `【考題】${dept.question}`;

     

      const optionsContainer = document.getElementById("modal-options");

      optionsContainer.innerHTML = "";

 

      dept.options.forEach((opt, optIdx) => {

        const btn = document.createElement("button");

        btn.className = "option-btn";

        btn.innerText = `${String.fromCharCode(65 + optIdx)}. ${opt}`;

        btn.onclick = () => checkAnswer(optIdx);

        optionsContainer.appendChild(btn);

      });

 

      const feedback = document.getElementById("modal-feedback");

      feedback.className = "feedback-box";

      feedback.style.display = "none";

 

      document.getElementById("modal-close").style.display = "none";

      document.getElementById("modal").classList.add("active");

    }

 

    // 檢查答案

    function checkAnswer(selectedIndex) {

      const dept = departmentsData[currentDeptIndex];

      const feedback = document.getElementById("modal-feedback");

      const optionBtns = document.querySelectorAll(".option-btn");

 

      optionBtns.forEach(btn => btn.style.pointerEvents = "none");

 

      if (selectedIndex === dept.answer) {

        playCorrectSound();

        feedback.className = "feedback-box correct";

        feedback.innerHTML = `<strong>答對了！</strong><br>${dept.explanation}`;

       

        if (!clearedDepts.has(currentDeptIndex)) {

          clearedDepts.add(currentDeptIndex);

          score += 100;

          document.getElementById(`card-${currentDeptIndex}`).classList.add("cleared");

          updateStats();

        }

      } else {

        playWrongSound();

        feedback.className = "feedback-box wrong";

        feedback.innerHTML = `<strong>遺憾答錯了！</strong><br>再試一次或者仔細思考司官的教誨喔！`;

      }

 

      document.getElementById("modal-close").style.display = "block";

    }

 

    // 關閉 Modal

    function closeModal() {

      document.getElementById("modal").classList.remove("active");

     

      if (clearedDepts.size === departmentsData.length) {

        setTimeout(() => {

          triggerGoldShower();

          alert(`🎉 恭喜你順利通過臺灣府城隍廟二十四司的所有考驗！\n總共獲得功德值：${score} 分！\n城隍爺賜予你「府城平安功德錦囊」，保佑你平安順遂！\n\n（製作：臺灣府城隍廟志工小V）`);

        }, 300);

      }

    }

 

    window.onload = initGame;
